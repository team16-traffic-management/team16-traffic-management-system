cmake_minimum_required(VERSION 3.16)
project(TrafficManagementSystem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tms_core src/TMS.cpp)
target_include_directories(tms_core PUBLIC include)
if(MSVC)
 target_compile_options(tms_core PRIVATE /W4)
else()
 target_compile_options(tms_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(tms src/main.cpp)
target_link_libraries(tms PRIVATE tms_core)
add_executable(tms_tests tests/test_tms.cpp)
target_link_libraries(tms_tests PRIVATE tms_core)
enable_testing()
add_test(NAME tms_tests COMMAND tms_tests)
